# Shaders sources separately
set(SHADER_DX11_VERTEX_INPUT    ${CMAKE_CURRENT_SOURCE_DIR}/shaders/dx11_vs_layering.hlsl)
set(SHADER_DX11_FRAGMENT_INPUT  ${CMAKE_CURRENT_SOURCE_DIR}/shaders/dx11_fs_layering.hlsl)
set(SHADER_DX12_VERTEX_INPUT    ${CMAKE_CURRENT_SOURCE_DIR}/shaders/dx12_vs_layering.hlsl)
set(SHADER_DX12_FRAGMENT_INPUT  ${CMAKE_CURRENT_SOURCE_DIR}/shaders/dx12_fs_layering.hlsl)
# Paths where the shaders should be compiled to
set(SHADER_DX11_VERTEX_OUTPUT   ${CMAKE_CURRENT_BINARY_DIR}/dx11_vs_layering.vs)
set(SHADER_DX11_FRAGMENT_OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/dx11_fs_layering.fs)
set(SHADER_DX12_VERTEX_OUTPUT   ${CMAKE_CURRENT_BINARY_DIR}/dx12_vs_layering.vs)
set(SHADER_DX12_FRAGMENT_OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/dx12_fs_layering.fs)

# Configure compile_shaders.bat script with above paths
configure_file(
    ${CMAKE_CURRENT_SOURCE_DIR}/templates/compile_shaders_template.bat
    ${CMAKE_CURRENT_SOURCE_DIR}/generated/compile_shaders.bat
)

# Set shader source files.
set(SHADERS
    ${SHADER_DX11_VERTEX_INPUT}
    ${SHADER_DX11_FRAGMENT_INPUT}
    ${SHADER_DX12_VERTEX_INPUT}
    ${SHADER_DX12_FRAGMENT_INPUT}
)

if(EMBED_SHADERS)
    message("Embedding shaders in library for release")
    # Compile shaders
    execute_process(COMMAND ${CMAKE_CURRENT_SOURCE_DIR}/generated/compile_shaders.bat)
    # Generate header file containing shader binary data
    file(MAKE_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/generated)
    FileToHexBuffer(${SHADER_DX11_VERTEX_OUTPUT}    "SHADER_DX11_VERTEX")
    FileToHexBuffer(${SHADER_DX11_FRAGMENT_OUTPUT}  "SHADER_DX11_FRAGMENT")
    FileToHexBuffer(${SHADER_DX12_VERTEX_OUTPUT}    "SHADER_DX12_VERTEX")
    FileToHexBuffer(${SHADER_DX12_FRAGMENT_OUTPUT}  "SHADER_DX12_FRAGMENT")
    configure_file(
        ${CMAKE_CURRENT_SOURCE_DIR}/templates/shaders_embedded_template.h
        ${CMAKE_CURRENT_SOURCE_DIR}/generated/shaders_generated.h

        )

    # Append generated files to library
    list(APPEND SHADERS
        generated/shaders_generated.h
        templates/shaders_embedded_template.h # Add only for easy editing
    )
endif()

set(ALL_LIBRARY_SOURCES
    ${SROUCES}
    ${HOTKEYS}
    ${GRAPHICS}
    ${SHADERS}
)

# Add target
add_library(RuntimeOpenXR SHARED
    ${ALL_LIBRARY_SOURCES}
)

if (EMBED_SHADERS)
    # Use embedded shaders
    target_compile_definitions(RuntimeOpenXR PRIVATE EMBED_SHADERS)
else()
    # Compile shaders as post build command
    add_custom_command(TARGET RuntimeOpenXR POST_BUILD
    COMMAND ${CMAKE_CURRENT_SOURCE_DIR}/generated/compile_shaders.bat
    )
endif()
